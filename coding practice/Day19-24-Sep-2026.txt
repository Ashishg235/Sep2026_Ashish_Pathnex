## Ansible - Deploy App Using Git

---
- name: Deploy app from Git for Pathnex
  hosts: all
  become: yes

  tasks:
    - name: Install Git
      yum:
        name: git
        state: present

    - name: Clone Repository
      git:
        repo: "https://github.com/pathnex/app.git"
        dest: "/opt/pathnex-app"

## Kubernetes - Deployment Rolling Update

apiVersion: apps/v1
kind: Deployment
metadata:
    name: pathnex-rollout
spec:
    replicas: 3
    strategy:
        type: RollingUpdate
        rollingUpdate:
            maxSurge: 2
            maxUnavailable: 1
    selector:
        matchLabels:
            app: rollout
    template:
        metadata:
            labels:
                app: rollout
        spec:
            containers:
                - name: app
                  image: nginx


## Shell Script - Check Internet Connectivity

#!/bin/bash

ping -c 2 google.com &> /dev/null

if [ $? -ep 0 ]; then
    echo "Internet is available"
else
    echo "No Internet connection"
fi


## Jenkins Pipeline - Mannual Approval

pipeline {
    ageny any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage('Deploy') {
            steps {
                input message: "Approve deployment for $INSTITUTE_NAME?"
                sh 'echo "Deployment approved and executed"'
            }
        }
    }
}
