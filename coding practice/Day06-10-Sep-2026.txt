## Ansible

- name: Install Nginx
  hosts: all
  become: yes
  
  tasks:
    - name: Install Nginx
  yum:
      name: nginx
      state: present

## Jenkins

pipeline {
	agent any
	stages {
		stage('Checkout') {
			steps {
				git branch: 'main', url: 'https://github.com/Pathnex/sample-java.app.git'
			}
		}
		stage('Build') {
			steps {
				sh '''
				mvn clean compile || { echo "Compile Failed"; exit 1; }
				'''
			}
		}
		stage('Test') {
			steps {
				sh '''
				mvn test || { echo "Test Failed"; exit 1; }
				'''
			}
		}
	}
}


## GitLab CI

stages:
   - build
   - test

build:
   stage: build
   image: maven:3.8.1-jdk-17
   script:
	- git clone https://github.com/Pathnex/sample-java-app.git
	- cd sample-java-app
	- mvn clean compile || { echo "Compile Failed"; exit 1; }

test:
   stage: test
   image: maven:3.8.1-jdk-17
   script: 
	- cd sample-java-app
	- mvn test || { echo "Test Failed"; exit 1; }
